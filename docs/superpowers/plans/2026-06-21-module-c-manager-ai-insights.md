# 模組 C：Manager 兩個 AI 分析功能 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 Manager 業務分析頁新增兩個 AI 功能：(A) 團隊整體診斷 + 逐業務點評；(B) 個別業務 coaching 輔導報告。皆「點按生成 + 存 DB 快取」，沿用既有 Spring AI + deterministic fallback + AiCallLog 治理。

**Architecture:** 後端新增 `ManagerInsightService`（同步呼叫 LLM，仿 `InsightService.portfolioAssessment`）+ `manager_insight` 快取表（scope=TEAM/OWNER）。AI-A 餵模組 B 的 `OwnerStats`，AI-B 餵該業務名下客戶 context（含模組 A 的 risk_level）。前端在統計頁加團隊診斷區塊（內嵌 markdown），業務表每列加「AI 建議」按鈕（重用既有 `ReportModal`）。

**Tech Stack:** Java 21、Spring Boot 4.1、Spring AI 2.0（`ChatClient`）、Flyway、JUnit 5 + Testcontainers；React 19 + react-markdown + remark-gfm。

**Scope note:** 整體 spec（`docs/superpowers/specs/2026-06-21-manager-sales-analytics-design.md`）三份計畫之第三份。**依賴模組 A**（`customers.risk_level` 欄位）與**模組 B**（`ManagerAnalyticsService`、`OwnerStats`、`/api/manager` RBAC、`ManagerAnalyticsPage`）。最新 migration 接續模組 A 的 V13 → 本計畫用 V14。

**設計決策：**
- **同步而非串流**：快取模式下進頁讀快取（秒回），只有「重新分析」打一次 LLM；同步實作清楚、無 Reactor 執行緒/交易陷阱，沿用 `portfolioAssessment` 成功範本。前端 AI 請求用既有 `AI_TIMEOUT`（120s）。
- **業務識別用 ownerName**：與模組 B 一致；`manager_insight.owner_key` 存 ownerName，TEAM scope 時為 null。
- **AI-B 重新生成（YAGNI）**：本期業務 coaching 為「點按查看（無快取則生成）」；團隊診斷提供「重新分析」按鈕。業務級重新分析留待後續，不在本期範圍。

```java
package com.aicrm.crm.domain;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.Table;
import java.time.Instant;

/**
 * Manager AI 洞察快取：避免每次進頁都重打 LLM。
 * scope=TEAM 時 ownerKey 為 null（團隊診斷）；scope=OWNER 時 ownerKey 為業務名（個別 coaching）。
 */
@Entity
@Table(name = "manager_insight")
public class ManagerInsight {

    /** 主鍵。 */
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    /** 範圍：TEAM / OWNER。 */
    @Column(nullable = false, length = 16)
    private String scope;

    /** 業務識別（ownerName）；TEAM 時為 null。 */
    @Column(name = "owner_key", length = 120)
    private String ownerKey;

    /** 報告內容（Markdown）。 */
    @Column(columnDefinition = "text", nullable = false)
    private String content;

    /** 產生此內容的模型（fallback 時為 null）。 */
    @Column(length = 120)
    private String model;

    /** 產生時間。 */
    @Column(name = "generated_at", nullable = false)
    private Instant generatedAt;

    protected ManagerInsight() {
    }

    /**
     * 建立洞察快取。
     *
     * @param scope TEAM / OWNER
     * @param ownerKey 業務名（TEAM 時 null）
     * @param content Markdown 內容
     * @param model 模型名稱（fallback 時 null）
     */
    public ManagerInsight(String scope, String ownerKey, String content, String model) {
        this.scope = scope;
        this.ownerKey = ownerKey;
        this.content = content;
        this.model = model;
        this.generatedAt = Instant.now();
    }

    /**
     * 更新內容（重新分析時 upsert 用）。
     *
     * @param content 新內容
     * @param model 模型名稱（fallback 時 null）
     */
    public void update(String content, String model) {
        this.content = content;
        this.model = model;
        this.generatedAt = Instant.now();
    }

    public Long getId() { return id; }
    public String getScope() { return scope; }
    public String getOwnerKey() { return ownerKey; }
    public String getContent() { return content; }
    public String getModel() { return model; }
    public Instant getGeneratedAt() { return generatedAt; }
}
```

```java
package com.aicrm.crm.service;

import com.aicrm.crm.api.Dtos;
import com.aicrm.crm.domain.AiCallType;
import com.aicrm.crm.domain.Customer;
import com.aicrm.crm.domain.Interaction;
import com.aicrm.crm.domain.ManagerInsight;
import com.aicrm.crm.domain.Opportunity;
import com.aicrm.crm.repository.CustomerRepository;
import com.aicrm.crm.repository.ManagerInsightRepository;
import java.math.BigDecimal;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Optional;
import java.util.function.Supplier;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.ai.chat.client.ChatClient;
import org.springframework.ai.chat.model.ChatModel;
import org.springframework.beans.factory.ObjectProvider;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

/**
 * Manager AI 洞察服務：產生「團隊整體診斷 + 逐業務點評」與「個別業務 coaching」報告，
 * 結果存 manager_insight 快取（點按生成）。沿用 InsightService 的可切換策略：
 * 有金鑰呼叫 Spring AI，否則 deterministic fallback；每次呼叫寫 AiCallLog 治理。
 */
@Service
@Transactional
public class ManagerInsightService {

    /** 記錄 LLM 呼叫失敗等事件。 */
    private static final Logger log = LoggerFactory.getLogger(ManagerInsightService.class);

    /** 業務主管教練系統提示詞，約束角色、語言與防幻覺。 */
    private static final String SYSTEM_PROMPT = """
            你是一位資深 B2B 銷售主管教練。請僅根據提供的「業務績效數據」「客戶資料」回答，
            不得編造數字或未經確認的承諾。回答需條理清楚、可執行，使用繁體中文。""";

    /** 業務分析聚合（提供 OwnerStats 給團隊診斷）。 */
    private final ManagerAnalyticsService analytics;

    /** 客戶存取（個別業務 coaching 組 context）。 */
    private final CustomerRepository customers;

    /** 洞察快取存取。 */
    private final ManagerInsightRepository insights;

    /** AI 治理：寫 ai_call_log。 */
    private final AiGovernanceService aiGovernance;

    /** ChatModel 提供者（Spring AI 2.0 即使無金鑰仍建立 bean）。 */
    private final ObjectProvider<ChatModel> chatModelProvider;

    /** 是否啟用真實 LLM：以 api-key 是否設定為準。 */
    private final boolean aiEnabled;

    public ManagerInsightService(ManagerAnalyticsService analytics,
                                 CustomerRepository customers,
                                 ManagerInsightRepository insights,
                                 AiGovernanceService aiGovernance,
                                 ObjectProvider<ChatModel> chatModelProvider,
                                 @Value("${spring.ai.openai.api-key:}") String openAiApiKey) {
        this.analytics = analytics;
        this.customers = customers;
        this.insights = insights;
        this.aiGovernance = aiGovernance;
        this.chatModelProvider = chatModelProvider;
        this.aiEnabled = openAiApiKey != null && !openAiApiKey.isBlank();
    }

    /** callLlm 回傳：內容與模型（fallback 時 model 為 null）。 */
    private record LlmOut(String content, String model) {}

    /**
     * 讀取團隊診斷快取。
     *
     * @return 團隊診斷（未產生則 empty）
     */
    @Transactional(readOnly = true)
    public Optional<Dtos.ManagerInsightResponse> getTeamInsight() {
        return insights.findByScopeAndOwnerKeyIsNull("TEAM").map(this::toResponse);
    }

    /**
     * 讀取個別業務 coaching 快取。
     *
     * @param ownerName 業務名
     * @return coaching（未產生則 empty）
     */
    @Transactional(readOnly = true)
    public Optional<Dtos.ManagerInsightResponse> getOwnerInsight(String ownerName) {
        return insights.findByScopeAndOwnerKey("OWNER", ownerName).map(this::toResponse);
    }

    /**
     * 產生（或重新產生）團隊整體診斷 + 逐業務點評，並 upsert 快取。
     *
     * @return 團隊診斷回應
     */
    public Dtos.ManagerInsightResponse generateTeamInsight() {
        var data = analytics.analytics();
        var prompt = buildTeamPrompt(data);
        var out = callLlm(AiCallType.TEAM_ANALYSIS, prompt, () -> deterministicTeam(data));
        return upsert("TEAM", null, out);
    }

    /**
     * 產生（或重新產生）個別業務 coaching，並 upsert 快取。
     *
     * @param ownerName 業務名
     * @return coaching 回應
     */
    public Dtos.ManagerInsightResponse generateOwnerInsight(String ownerName) {
        var ownerCustomers = customers.findByOwnerName(ownerName);
        var prompt = buildOwnerPrompt(ownerName, ownerCustomers);
        var out = callLlm(AiCallType.OWNER_COACHING, prompt, () -> deterministicOwner(ownerName, ownerCustomers));
        return upsert("OWNER", ownerName, out);
    }

    /**
     * 呼叫 LLM（同步）；無金鑰、空白或失敗皆走 fallback。每次都寫 ai_call_log。
     *
     * @param type 呼叫類型
     * @param userPrompt 使用者提示（已含資料，送出前過 PII 遮罩）
     * @param fallback deterministic 後備內容供應器
     * @return 內容與模型
     */
    private LlmOut callLlm(AiCallType type, String userPrompt, Supplier<String> fallback) {
        var chatModel = aiEnabled ? chatModelProvider.getIfAvailable() : null;
        if (chatModel == null) {
            var answer = fallback.get();
            aiGovernance.record(type, null, null, 0, 0, 0, false, true, answer);
            return new LlmOut(answer, null);
        }
        try {
            var resp = ChatClient.create(chatModel).prompt()
                    .system(SYSTEM_PROMPT)
                    .user(PiiMasker.mask(userPrompt))
                    .call().chatResponse();
            var content = resp == null ? null : resp.getResult().getOutput().getText();
            if (content == null || content.isBlank()) {
                var answer = fallback.get();
                aiGovernance.record(type, null, null, 0, 0, 0, false, true, answer);
                return new LlmOut(answer, null);
            }
            var metadata = resp.getMetadata();
            var usage = metadata == null ? null : metadata.getUsage();
            var model = metadata == null ? null : metadata.getModel();
            Integer pt = usage == null ? null : usage.getPromptTokens();
            Integer ct = usage == null ? null : usage.getCompletionTokens();
            Integer tt = usage == null ? null : usage.getTotalTokens();
            var answer = content.strip();
            aiGovernance.record(type, null, model, pt, ct, tt, true, true, answer);
            return new LlmOut(answer, model);
        } catch (Exception e) {
            log.warn("Manager AI 洞察呼叫失敗，type={}，改用 fallback：{}", type, e.getMessage());
            var answer = fallback.get();
            aiGovernance.record(type, null, null, 0, 0, 0, false, true, answer);
            return new LlmOut(answer, null);
        }
    }

    /**
     * Upsert 快取並回傳 DTO。
     */
    private Dtos.ManagerInsightResponse upsert(String scope, String ownerKey, LlmOut out) {
        var existing = ownerKey == null
                ? insights.findByScopeAndOwnerKeyIsNull(scope)
                : insights.findByScopeAndOwnerKey(scope, ownerKey);
        ManagerInsight entity = existing.orElseGet(() -> new ManagerInsight(scope, ownerKey, out.content(), out.model()));
        if (existing.isPresent()) {
            entity.update(out.content(), out.model());
        }
        var saved = insights.save(entity);
        return toResponse(saved);
    }

    /** entity → DTO。 */
    private Dtos.ManagerInsightResponse toResponse(ManagerInsight m) {
        return new Dtos.ManagerInsightResponse(m.getScope(), m.getOwnerKey(), m.getContent(), m.getModel(), m.getGeneratedAt());
    }

    /**
     * 組裝團隊診斷提示：各業務 OwnerStats 表 + 要求整體診斷與逐人點評。
     */
    private String buildTeamPrompt(Dtos.ManagerAnalyticsResponse data) {
        var rows = data.owners().stream()
                .map(o -> o.ownerName() + "｜客戶 " + o.customerCount()
                        + "｜成交 " + o.wonAmount() + "(" + o.wonCount() + "件)"
                        + "｜成交率 " + Math.round(o.winRate() * 100) + "%"
                        + "｜Pipeline " + o.pipelineAmount() + "(" + o.activeOpportunityCount() + "案)"
                        + "｜高風險 " + o.highRiskCount()
                        + "｜本季續約 " + o.renewalsThisQuarter())
                .collect(java.util.stream.Collectors.joining("\n"));
        return """
                以下是銷售團隊各業務的績效數據（系統計算，請勿更改數字）：

                # 團隊總覽
                業務人數：%d｜團隊成交額：%s｜團隊 Pipeline：%s｜平均成交率：%d%%｜高風險客戶：%d

                # 各業務績效（業務名｜客戶數｜成交｜成交率｜Pipeline｜高風險｜本季續約）
                %s

                # 報告要求
                請產出「團隊業務分析報告」（Markdown）：
                1. **團隊整體診斷**：top 表現者、落後者、共通問題與機會。
                2. **逐業務點評**：對每一位業務各一段，列出「優勢」與「加強建議」。
                使用繁體中文、條理清楚、勿編造數字。
                """.formatted(
                data.team().ownerCount(), data.team().totalWonAmount(), data.team().totalPipeline(),
                Math.round(data.team().avgWinRate() * 100), data.team().totalHighRisk(), rows);
    }

    /**
     * 組裝個別業務 coaching 提示：該業務客戶清單 + 要求輔導建議。
     */
    private String buildOwnerPrompt(String ownerName, List<Customer> ownerCustomers) {
        var rows = ownerCustomers.stream()
                .map(c -> {
                    var amount = c.getOpportunities().stream().map(Opportunity::getAmount).reduce(BigDecimal.ZERO, BigDecimal::add);
                    var last = c.getInteractions().stream().map(Interaction::getOccurredAt).max(LocalDateTime::compareTo)
                            .map(d -> d.toLocalDate().toString()).orElse("無");
                    var risk = c.getRiskLevel() == null ? "未評" : c.getRiskLevel();
                    return "- " + c.getName() + "｜" + c.getIndustry() + "｜風險 " + risk
                            + "｜商機 " + c.getOpportunities().size() + "筆/" + amount
                            + "｜最近互動 " + last
                            + "｜續約日 " + (c.getRenewalDueDate() == null ? "未定" : c.getRenewalDueDate());
                })
                .collect(java.util.stream.Collectors.joining("\n"));
        return """
                以下是業務「%s」名下所有客戶的資料（系統計算，請勿更改數字）：

                # 客戶清單（名稱｜產業｜風險｜商機｜最近互動｜續約日）
                %s

                # 報告要求
                請以銷售主管視角，為這位業務產出「coaching 輔導報告」（Markdown）：
                1. **整體表現摘要**。
                2. **該優先跟進的客戶**（附理由，點名客戶）。
                3. **需立即處理的風險**（高風險或續約逾期客戶）。
                4. **具體輔導建議**（可執行）。
                使用繁體中文、條理清楚、勿編造價格或承諾。
                """.formatted(ownerName, rows.isBlank() ? "（此業務目前無客戶）" : rows);
    }

    /**
     * 團隊診斷 deterministic fallback。
     */
    private String deterministicTeam(Dtos.ManagerAnalyticsResponse data) {
        var top = data.owners().isEmpty() ? "（無資料）" : data.owners().get(0).ownerName();
        return """
                ## 團隊業務分析（教學版摘要）

                - **業務人數**：%d
                - **團隊成交額**：%s
                - **團隊 Pipeline**：%s
                - **平均成交率**：%d%%
                - **高風險客戶數**：%d
                - **成交額領先業務**：%s

                > 目前未設定 OpenAI 金鑰，顯示為系統彙總的 deterministic 摘要；設定金鑰後可取得 AI 團隊診斷與逐業務點評。
                """.formatted(
                data.team().ownerCount(), data.team().totalWonAmount(), data.team().totalPipeline(),
                Math.round(data.team().avgWinRate() * 100), data.team().totalHighRisk(), top);
    }

    /**
     * 個別業務 coaching deterministic fallback。
     */
    private String deterministicOwner(String ownerName, List<Customer> ownerCustomers) {
        long high = ownerCustomers.stream().filter(c -> "HIGH".equals(c.getRiskLevel())).count();
        return """
                ## %s 的業務輔導摘要（教學版）

                - **負責客戶數**：%d
                - **高風險客戶數**：%d

                > 目前未設定 OpenAI 金鑰，顯示為系統彙總的 deterministic 摘要；設定金鑰後可取得 AI 個別業務 coaching 輔導建議。
                """.formatted(ownerName, ownerCustomers.size(), high);
    }
}
```
